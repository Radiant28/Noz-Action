cmake_minimum_required(VERSION 3.20)
project(NozdorAction LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use an x86 Windows compiler; this client is 32 bit")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
add_compile_options(-Wall -Wextra -Werror)
add_library(NozdorAction SHARED src/native.cpp)
set_target_properties(NozdorAction PROPERTIES PREFIX "")
target_compile_options(NozdorAction PRIVATE -fno-exceptions -fno-rtti)
target_link_libraries(NozdorAction PRIVATE user32)
add_executable(NozdorActionLauncher src/launcher.cpp)
target_link_options(NozdorActionLauncher PRIVATE -municode)
target_link_libraries(NozdorActionLauncher PRIVATE advapi32 user32)
foreach(target NozdorAction NozdorActionLauncher)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
endforeach()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_compile_options(core_tests PRIVATE -UNDEBUG)
target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
add_test(NAME core COMMAND core_tests)
add_executable(native_interaction_tests tests/native_interaction_tests.cpp)
target_compile_options(native_interaction_tests PRIVATE -UNDEBUG)
target_link_options(native_interaction_tests PRIVATE -static -static-libgcc -static-libstdc++)
target_link_libraries(native_interaction_tests PRIVATE user32)
add_test(NAME interaction_capture COMMAND native_interaction_tests)
add_executable(lua_runner tests/lua_runner.c)
add_test(NAME addon COMMAND lua_runner "${CMAKE_SOURCE_DIR}/tests/addon_test.lua" "${CMAKE_SOURCE_DIR}/addon/NozdorAction/NozdorAction.lua")
add_test(NAME addon_ru COMMAND lua_runner "${CMAKE_SOURCE_DIR}/tests/addon_test.lua" "${CMAKE_SOURCE_DIR}/addon/NozdorAction/NozdorAction.lua" ruRU)
add_test(NAME addon_fallback COMMAND lua_runner "${CMAKE_SOURCE_DIR}/tests/addon_test.lua" "${CMAKE_SOURCE_DIR}/addon/NozdorAction/NozdorAction.lua" deDE)
if(EXISTS "${CMAKE_SOURCE_DIR}/../Interface/AddOns/!!RefinedBlizzPlates/Functions.lua")
    add_test(NAME refined_class_colors COMMAND lua_runner "${CMAKE_SOURCE_DIR}/tests/rbp_compat_test.lua" "${CMAKE_SOURCE_DIR}/../Interface/AddOns/!!RefinedBlizzPlates/Functions.lua")
endif()
add_test(NAME client_check COMMAND NozdorActionLauncher --check)
add_test(NAME reject_wrong_client COMMAND NozdorActionLauncher "$<TARGET_FILE:core_tests>" --check)
set_tests_properties(reject_wrong_client PROPERTIES WILL_FAIL TRUE)
add_test(NAME reject_foreign_host COMMAND core_tests "$<TARGET_FILE:NozdorAction>")
add_library(loader_fixture SHARED tests/loader_fixture.cpp)
set_target_properties(loader_fixture PROPERTIES PREFIX "")
add_executable(loader_child tests/loader_child.cpp)
add_executable(loader_smoke tests/loader_smoke.cpp)
target_compile_options(loader_smoke PRIVATE -UNDEBUG)
target_link_options(loader_smoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
target_link_libraries(loader_smoke PRIVATE advapi32 user32)
add_test(NAME suspended_loader COMMAND loader_smoke "$<TARGET_FILE:loader_child>" "$<TARGET_FILE:loader_fixture>")
